  130  private boolean isValidInput() {
  131    return isValidProviderName(providerNameText.getText().trim()) && isValidEndpoint(endpointText.getText().trim());
  132  }
  133 
  134  /**
  135   * The language server treats any name that is not a built-in provider as a custom endpoint provider, so built-in
  136   * names must be rejected here.
  137   */ 
  138  private boolean isValidProviderName(String name) {
  139    return StringUtils.isNotBlank(name) && ByokModelProvider.isCustomProvider(name);
  140  }

  142  private boolean isValidEndpoint(String value) {
  143    try {
  144      URI uri = new URI(value);
  145      return ("http".equalsIgnoreCase(uri.getScheme()) || "https".equalsIgnoreCase(uri.getScheme()))
  146          && StringUtils.isNotBlank(uri.getHost());
  147    } catch (URISyntaxException e) {
  148      return false;
  149    }
  150  }
